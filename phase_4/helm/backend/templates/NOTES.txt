Backend deployment successful!

1. Get the backend URL:
{{- if eq .Values.service.type "NodePort" }}
   export NODE_PORT=$(kubectl get --namespace {{ .Release.Namespace }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ include "backend.fullname" . }})
   export NODE_IP=$(kubectl get nodes --namespace {{ .Release.Namespace }} -o jsonpath="{.items[0].status.addresses[0].address}")
   echo http://$NODE_IP:$NODE_PORT

   Or use Minikube:
   minikube service {{ include "backend.fullname" . }} --url
{{- else if eq .Values.service.type "ClusterIP" }}
   kubectl port-forward svc/{{ include "backend.fullname" . }} {{ .Values.service.port }}:{{ .Values.service.port }}
{{- end }}

2. Check health endpoint:
   curl $(minikube service {{ include "backend.fullname" . }} --url)/health

3. View logs:
   kubectl logs -l app=backend

4. Check pod status:
   kubectl get pods -l app=backend
